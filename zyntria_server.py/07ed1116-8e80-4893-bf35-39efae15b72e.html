Future sendToLaptop(String command) async {
setState(() => lastAnswer = "Thinking...");
try {
final url = Uri.parse('http://$laptopIP:5000/command');
final res = await http.post(url, headers: {"Content-Type": "application/json"}, body: jsonEncode({"command": command}));
final data = jsonDecode(res.body);
setState(() => lastAnswer = data['result']);
await _tts.speak(data['result']);
} catch (e) {
setState(() => lastAnswer = "Error: Laptop not connected. Check IP.");
}
}

void startListening() async {
bool available = await _speech.initialize();
if (available) {
setState(() => isListening = true);
_speech.listen(onResult: (result) => _controller.text = result.recognizedWords);
}
}

void stopListening() {
_speech.stop();
setState(() => isListening = false);
if (_controller.text.isNotEmpty) sendToLaptop(_controller.text);
}
